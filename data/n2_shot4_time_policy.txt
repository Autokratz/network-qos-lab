===== NETWORK 2 : TIME-BASED BANDWIDTH THROTTLING (CLOUD BACKUP) =====
Uplink 20 Mbps.  Office-hours cap for backup host 10.20.0.50 = 1 Mbps (5%)

BR-RTR-01# nft list ruleset
table inet BRANCH_WAN_POLICY {
	set BACKUP_HOSTS {
		type ipv4_addr
		comment "cloud-backup agents at the branch site"
		elements = { 10.20.0.50 }
	}

	chain FORWARD {
		type filter hook forward priority filter; policy accept;
		ip saddr @BACKUP_HOSTS ip daddr 10.30.0.0/30 meta hour "09:00"-"17:00" limit rate over 125 kbytes/second burst 20 kbytes counter packets 0 bytes 0 drop comment "BACKUP-CAP-OFFICE-HOURS-5PCT"
		ip saddr @BACKUP_HOSTS ip daddr 10.30.0.0/30 limit rate over 125 kbytes/second burst 20 kbytes counter packets 0 bytes 0 drop comment "BACKUP-CAP-LAB-VERIFY-5PCT"
		ip saddr 10.20.0.0/24 ip daddr 10.30.0.0/30 counter packets 0 bytes 0 comment "BUSINESS-TRAFFIC-PERMIT"
	}
}
